function doGet(e) { var params = e.parameter || {}; var callback = params.callback || "handleGASResponse"; try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var result = { success: true, Registros: leerHojaComoObjetos(ss.getSheetByName("Documentos")), Catalogos: { Personal: leerHojaComoObjetos(ss.getSheetByName("Personal")), Estados: leerHojaComoObjetos(ss.getSheetByName("Estados")), Origenes: leerHojaComoObjetos(ss.getSheetByName("Origenes")), Alarmas: leerHojaComoObjetos(ss.getSheetByName("Alarmas")) } }; var output = callback + "(" + JSON.stringify(result) + ");"; return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JAVASCRIPT); } catch (err) { var errorResult = { success: false, message: "Error: " + err.toString() }; return ContentService.createTextOutput(callback + "(" + JSON.stringify(errorResult) + ");").setMimeType(ContentService.MimeType.JAVASCRIPT); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = data.action; if (action === "saveDoc") { var sheet = ss.getSheetByName("Documentos"); var rows = sheet.getDataRange().getValues(); var numDoc = String(data.Numero_Documento || "").trim(); var foundRow = -1; for (var i = 1; i < rows.length; i++) { if (String(rows[i][1]).trim().toLowerCase() === numDoc.toLowerCase()) { foundRow = i + 1; break; } } var urlsIngreso = foundRow > 0 ? (rows[foundRow-1][11] || "") : ""; if (data.Archivos_Ingreso && data.Archivos_Ingreso.length > 0) { var subidasIng = guardarArchivosEnDrive(data.Archivos_Ingreso, numDoc, "SUSTENTO"); if (subidasIng) urlsIngreso = urlsIngreso ? urlsIngreso + ", " + subidasIng : subidasIng; } var urlsCierre = foundRow > 0 ? (rows[foundRow-1][12] || "") : ""; if (data.Archivos_Cierre && data.Archivos_Cierre.length > 0) { var subidasCierre = guardarArchivosEnDrive(data.Archivos_Cierre, numDoc, "CIERRE"); if (subidasCierre) urlsCierre = urlsCierre ? urlsCierre + ", " + subidasCierre : subidasCierre; } var rowData = [ foundRow > 0 ? rows[foundRow-1][0] : rows.length, data.Numero_Documento || "", data.Origen_Tipo || "", data.Asunto || "", data.Fecha_Ingreso || "", data.Fecha_Vencimiento || "", data.Dias_Trancurridos || 0, data.Dias_Exceso || 0, data.Responsable || "", data.Estado || "", data.Observaciones || "", urlsIngreso, urlsCierre ]; if (foundRow > 0) { sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]); } else { sheet.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } if (action === "deleteDoc") { var sheet = ss.getSheetByName("Documentos"); var rows = sheet.getDataRange().getValues(); var numDoc = String(data.Numero_Documento || "").trim(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][1]).trim().toLowerCase() === numDoc.toLowerCase()) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } if (action === "saveCatalog") { var sheetName = data.type; var sheet = ss.getSheetByName(sheetName); if (!sheet) throw "Hoja de catálogo no encontrada: " + sheetName; var rows = sheet.getDataRange().getValues(); var val = String(data.value || "").trim(); var oldValue = String(data.oldValue || val).trim(); var foundRow = -1; for (var i = 1; i < rows.length; i++) { var colVal = String(rows[i][1] !== undefined ? rows[i][1] : rows[i][0]).trim(); if (colVal.toLowerCase() === oldValue.toLowerCase()) { foundRow = i + 1; break; } } var nextId = rows.length > 0 ? rows.length : 1; if (foundRow > 0) nextId = rows[foundRow-1][0]; var catalogRowData = []; if (sheetName === "Personal") { catalogRowData = [nextId, val, data.secondary || "", data.third || "", data.usuario || "", data.clave || "", data.tipo_usuario || "Operador"]; } else if (sheetName === "Estados") { catalogRowData = [nextId, val, data.secondary || "#3b82f6"]; } else if (sheetName === "Origenes") { catalogRowData = [nextId, val, data.secondary || ""]; } else if (sheetName === "Alarmas") { catalogRowData = [nextId, val, data.secondary || 0, data.third || 15, data.fourth || "#ef4444"]; } if (foundRow > 0) { sheet.getRange(foundRow, 1, 1, catalogRowData.length).setValues([catalogRowData]); } else { sheet.appendRow(catalogRowData); } return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } if (action === "deleteCatalog") { var sheetName = data.type; var sheet = ss.getSheetByName(sheetName); var rows = sheet.getDataRange().getValues(); var val = String(data.value || "").trim(); for (var i = 1; i < rows.length; i++) { var colVal = String(rows[i][1] !== undefined ? rows[i][1] : rows[i][0]).trim(); if (colVal.toLowerCase() === val.toLowerCase()) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } } catch(err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function leerHojaComoObjetos(sheet) { if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length < 2) return []; var headers = data[0]; var result = []; for (var i = 1; i < data.length; i++) { var row = data[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { var headerName = String(headers[j]).trim(); var val = row[j]; if (val instanceof Date) { val = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } obj[headerName] = val; obj[j] = val; } result.push(obj); } return result; } function guardarArchivosEnDrive(archivosBase64, numeroDocumento, tipoCarpeta) { if (!archivosBase64 || archivosBase64.length === 0) return ""; try { var folders = DriveApp.getFoldersByName("BD_Documentos_Ventas"); var carpetaPrincipal = folders.hasNext() ? folders.next() : DriveApp.createFolder("BD_Documentos_Ventas"); var subFolders = carpetaPrincipal.getFoldersByName("Archivos"); var carpetaArchivos = subFolders.hasNext() ? subFolders.next() : carpetaPrincipal.createFolder("Archivos"); var docFolders = carpetaArchivos.getFoldersByName(numeroDocumento); var carpetaDoc = docFolders.hasNext() ? docFolders.next() : carpetaArchivos.createFolder(numeroDocumento); var urls = []; for (var i = 0; i < archivosBase64.length; i++) { var item = archivosBase64[i]; try { var blob = Utilities.newBlob(Utilities.base64Decode(item.data), item.mimeType, tipoCarpeta + "_" + item.name); var archivoDrive = carpetaDoc.createFile(blob); archivoDrive.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); urls.push(archivoDrive.getUrl()); } catch(e) {} } return urls.join(", "); } catch(err) { return ""; } }
